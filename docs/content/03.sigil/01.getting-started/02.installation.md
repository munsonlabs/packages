### Install

```bash
npm install @munsonlabs/sigil
```

No dependencies. `vue` is optional, for the `/vue` entry.

### Register a library

```ts
import { register } from '@munsonlabs/sigil'

register(
  'lucide',
  {
    resolver: (name) => `https://cdn.jsdelivr.net/npm/lucide-static/icons/${name}.svg`,
    mutator: (svg) => svg.setAttribute('stroke', 'currentColor'),
  },
  { default: true },
)
```

### Render

::tabs
:::tabs-item{label="Web component" icon="i-vscode-icons:file-type-html"}

```ts
import '@munsonlabs/sigil/element'
```

```html
<ml-sigil-icon name="heart"></ml-sigil-icon>
```

:::
:::tabs-item{label="Vue" icon="i-vscode-icons:file-type-vue"}

```vue
<script setup>
import { Sigil } from '@munsonlabs/sigil/vue'
</script>

<template>
  <Sigil name="heart" />
</template>
```

:::
:::tabs-item{label="No build step" icon="i-lucide:globe"}

```html
<script type="module">
  import { register } from 'https://esm.sh/@munsonlabs/sigil'
  import 'https://esm.sh/@munsonlabs/sigil/element'

  register(
    'lucide',
    {
      resolver: (name) => `https://cdn.jsdelivr.net/npm/lucide-static/icons/${name}.svg`,
    },
    { default: true },
  )
</script>

<ml-sigil-icon name="heart"></ml-sigil-icon>
```

No import map needed.
:::
::

### Size it

```css
ml-sigil-icon {
  display: inline-flex;
  width: 1em;
  height: 1em;
}
ml-sigil-icon svg {
  width: 100%;
  height: 100%;
}
```

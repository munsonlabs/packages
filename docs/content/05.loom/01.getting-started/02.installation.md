## 1. Make a kit

A kit is everything your widgets can read. Start with one source wrapping a composable you already have:

```js [kit.js]
import { createKit, defineSource } from '@munsonlabs/loom'
import { useProduct } from './useProduct'

export const kit = createKit({
  sources: [
    defineSource({
      id: 'product',
      label: 'Product',
      fields: {
        name: { type: 'string' },
        price: { type: 'string' },
        soldOut: { type: 'boolean' },
      },
      setup: (ctx) =>
        useProduct(
          {
            get product() {
              return ctx.data
            },
          },
          ctx.context.cart,
        ),
    }),
  ],
})
```

## 2. Render a widget

```vue
<script setup>
import { Widget, provideKit } from '@munsonlabs/loom'
import '@munsonlabs/loom/style.css'
import { kit } from './kit'
import productCard from './product-card.widget.json'

provideKit(kit, shop) // the kit, and the context its sources get
</script>

<template>
  <Widget v-for="(product, i) in products" :key="product.id" :definition="productCard" :data="product" :index="i" />
</template>
```
